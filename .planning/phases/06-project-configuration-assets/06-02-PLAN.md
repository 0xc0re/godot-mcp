```typescript
export function registerCompositionTools(server: McpServer, ctx: ServerContext): void {
  server.registerTool('tool_name', {
    title: 'Tool Title',
    description: 'Description',
    inputSchema: { project_path: z.string(), /* ... */ },
  }, async ({ project_path, /* ... */ }) => {
    if (!validatePath(project_path)) return toolError('Invalid path', [...]);
    try {
      const projectFile = join(project_path, 'project.godot');
      if (!existsSync(projectFile)) return toolError('Not a valid Godot project: ...', [...]);
      const params = { camelCaseKey: snake_case_input };
      const { stdout, stderr } = await executeOperation(ctx, project_path, 'operation_name', params);
      if (stderr && (stderr.includes('Failed to') || stderr.includes('[ERROR]'))) return toolError(...);
      return { content: [{ type: 'text' as const, text: stdout }] };
    } catch (error: unknown) {
      const errorMessage = error instanceof Error ? error.message : 'Unknown error';
      return toolError(`Failed to ...: ${errorMessage}`, [...]);
    }
  });
}
```
