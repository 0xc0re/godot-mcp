import { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js';
import { z } from 'zod';
import { join } from 'path';
import { existsSync } from 'fs';
import type { ServerContext } from '../types.js';
import { executeOperation, validatePath } from '../godot.js';
import { toolError } from '../errors.js';

export function registerCompositionTools(server: McpServer, ctx: ServerContext): void {
  server.registerTool(
    'tool_name',
    {
      title: 'Tool Title',
      description: 'Tool description...',
      inputSchema: {
        project_path: z.string().describe('Path to the Godot project directory'),
        // ... more params with z schemas ...
      },
    },
    async ({ project_path, ...params }) => {
      if (!validatePath(project_path) || !validatePath(other_path)) {
        return toolError('Invalid path', [
          'Provide valid paths without ".." or other potentially unsafe characters',
        ]);
      }
      if (!existsSync(join(project_path, 'project.godot'))) {
        return toolError('project.godot not found', [
          'Provide a valid Godot project directory',
        ]);
      }
      try {
        const { stdout, stderr } = await executeOperation(ctx, project_path, 'operation_name', {
          camelCaseParam: snake_case_input,
        });
        if (stderr && stderr.includes('Failed to')) {
          return toolError(`Operation failed: ${stderr}`);
        }
        return { content: [{ type: 'text' as const, text: stdout }] };
      } catch (error: unknown) {
        const errorMessage = error instanceof Error ? error.message : 'Unknown error';
        return toolError(`Failed: ${errorMessage}`, ['Check Godot installation', ...]);
      }
    },
  );
}
